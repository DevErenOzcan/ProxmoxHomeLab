# =============================================================================
# The Python environment of the CONTROLLER - the machine that runs
# ansible-playbook - not of the managed hosts.
# =============================================================================
# The controller is the repo-root virtualenv, ProxmoxHomeLab/.venv, created
# from WSL/Linux (ansible has no Windows control node). Fill it from the repo
# root with:
#
#   .venv/bin/pip install -r ansible/requirements.txt
#   .venv/bin/ansible-galaxy collection install -r ansible/requirements.yml
#
# The venv owns ansible itself, not only these libraries: apt's
# /usr/bin/ansible starts with the absolute shebang "#!/usr/bin/python3" and
# imports from /usr/lib/python3/dist-packages, where nothing installed here
# exists - httpx included. Activating the venv (or calling
# .venv/bin/ansible-playbook) is what selects this environment.

# Pinned on purpose: ansible-core 2.19 replaced the template engine, so an
# unpinned upgrade can change how the roles' conditionals and asserts evaluate.
# Upgrade deliberately, then re-run --syntax-check and a --check of each
# playbook.
ansible-core==2.21.4

# What the ansibleguy.opnsense modules talk to the OPNsense REST API with. They
# run on the controller (playbooks/opnsense/site.yml uses connection: local),
# so httpx has to exist in this environment.
#
# The floor is 0.28 deliberately: older stacks (httpx 0.23 / httpcore 0.15 /
# h11 0.12) mis-report a broken server response as "illegal chunk header";
# 0.28 says "malformed chunk footer", which is what happens. That error is not
# a client bug: OPNsense 26.7.0 corrupted HTTP/1.1 chunked responses past
# ~64 KiB (GET /api/firewall/filter/get, which the rule module reads first);
# 26.7.1 fixed it. Keep the firewall on 26.7.1 or later.
httpx>=0.28.0
